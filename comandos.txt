#### Comando usados na tentativa de migração de jobs existentes...

#### O comando usado abaixo tem como objetivo automatizar o YML de um job existente no workspace


$jobs = databricks jobs list --output json | ConvertFrom-Json 

$filtered = $jobs | Where-Object { $_.settings.name -match '(?i)\[DTCORE\]' }

foreach ($job in $filtered) {
    $job_id   = $job.job_id
    $job_name = $job.settings.name

    databricks bundle generate job --existing-job-id $job_id --bind  --config-dir "resources/jobs/migrados" --source-dir "resources/notebooks/migrados" --force

    Write-Host "OK  -> $job_id | $job_name"
}
